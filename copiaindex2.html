function ajustarTelaReferencia() { const larguraReferencia = 1920; const alturaReferencia = 1080; const celular = window.matchMedia("(pointer: coarse)").matches || window.innerWidth <= 600; const viewport = celular ? window.visualViewport : null; const larguraJanela = viewport ? viewport.width : window.innerWidth; const alturaJanela = viewport ? viewport.height : window.innerHeight; const deslocamentoEsquerda = viewport ? viewport.offsetLeft : 0; const deslocamentoTopo = viewport ? viewport.offsetTop : 0; const celularVertical = celular && alturaJanela >= larguraJanela; const celularHorizontal = celular && larguraJanela > alturaJanela; let enquadramento = { zoom: 1, horizontal: 0.5, vertical: 0.5 }; if (celularVertical) { enquadramento = { zoom: 1.05, horizontal: 0.5, vertical: 0 }; } else if (celularHorizontal) { enquadramento = { zoom: 1, horizontal: 0.5, vertical: 0 }; } const escalaBase = Math.max( larguraJanela / larguraReferencia, alturaJanela / alturaReferencia ); const escala = escalaBase * enquadramento.zoom; const larguraEscalada = larguraReferencia * escala; const alturaEscalada = alturaReferencia * escala; telaReferencia.style.transform = `scale(${escala})`; telaReferencia.style.left = `${deslocamentoEsquerda + (larguraJanela - larguraEscalada) * enquadramento.horizontal}px`; telaReferencia.style.top = `${deslocamentoTopo + (alturaJanela - alturaEscalada) * enquadramento.vertical}px`; } ajustarTelaReferencia(); window.addEventListener("resize", ajustarTelaReferencia); window.addEventListener("orientationchange", ajustarTelaReferencia); if (window.visualViewport) { window.visualViewport.addEventListener( "resize", ajustarTelaReferencia ); window.visualViewport.addEventListener( "scroll", ajustarTelaReferencia ); } /* ELEMENTOS DO SITE */
